1. APLICAR E VALIDAR
--------------------

$repo = "C:\Users\TH\Documents\GitHub\estoque-superinfra"
$pacote = "$HOME\Downloads\atualizacao-66-historico-retorno-tecnico.zip"
$aplicador = "$HOME\Downloads\APLICAR-ATUALIZACAO-66.ps1"

if (!(Test-Path -LiteralPath $repo -PathType Container)) {
    throw "Projeto não encontrado em: $repo"
}

if (!(Test-Path -LiteralPath $pacote -PathType Leaf)) {
    throw "Pacote não encontrado em: $pacote"
}

if (!(Test-Path -LiteralPath $aplicador -PathType Leaf)) {
    throw "Aplicador não encontrado em: $aplicador"
}

Unblock-File -LiteralPath $aplicador

Set-ExecutionPolicy `
    -Scope Process `
    -ExecutionPolicy Bypass `
    -Force

& $aplicador `
    -Repo $repo `
    -Zip $pacote

2. ENVIAR AO GITHUB
-------------------

Set-Location "C:\Users\TH\Documents\GitHub\estoque-superinfra"

$arquivos = @(
    "backend/app/controllers/stockController.js",
    "backend/app/routes/stockRoutes.js",
    "frontend/src/pages/TechnicianReturns.jsx",
    "ATUALIZACAO-66-HISTORICO-RETORNO-TECNICO.md",
    "RELATORIO-VALIDACAO-ATUALIZACAO-66.txt",
    "COMANDOS-ATUALIZACAO-66-POWERSHELL.txt",
    "APLICAR-ATUALIZACAO-66.ps1"
)

git add -- $arquivos

git diff --cached --check

if ($LASTEXITCODE -ne 0) {
    throw "O Git encontrou problemas. Não faça o commit."
}

git status --short

git commit -m "Exibe histórico na página de retorno técnico"

$branch = (git branch --show-current).Trim()

if ([string]::IsNullOrWhiteSpace($branch)) {
    throw "Não foi possível identificar a branch atual."
}

git push origin $branch
